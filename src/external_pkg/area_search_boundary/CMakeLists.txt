cmake_minimum_required(VERSION 3.8)
project(area_search_boundary)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/boundary_layer.cpp)
ament_target_dependencies(${PROJECT_NAME} rclcpp nav2_costmap_2d pluginlib geometry_msgs std_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()
