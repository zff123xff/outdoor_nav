cmake_minimum_required(VERSION 3.16)
project(co_search_nav)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(nlohmann_json REQUIRED)
set(dependencies exploration_manager plan_env plan_manage path_searching active_perception bspline bspline_opt traj_utils poly_traj ppu_opt snn_region quadrotor_msgs msg_set lkh_tsp_solver lkh_mtsp_solver rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros tf2_geometry_msgs pcl_ros pcl_conversions cv_bridge visualization_msgs)
foreach(dep ${dependencies})
  find_package(${dep} REQUIRED)
endforeach()
# Compile the original manager unchanged; only ROS I/O and execution live here.
add_executable(search_engine src/search_engine.cpp ../upstream/exploration_manager/src/fast_exploration_manager.cpp)
target_include_directories(search_engine PRIVATE ../upstream/exploration_manager/include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
ament_target_dependencies(search_engine ${dependencies})
target_link_libraries(search_engine ${PCL_LIBRARIES} nlohmann_json::nlohmann_json m)
include(lkh_sources.cmake)
add_executable(co_search_lkh ${CO_SEARCH_LKH_SOURCES})
target_include_directories(co_search_lkh PRIVATE vendor/LKH/INCLUDE)
target_compile_definitions(co_search_lkh PRIVATE TWO_LEVEL_TREE)
target_compile_options(co_search_lkh PRIVATE -w)
target_link_libraries(co_search_lkh m)
install(TARGETS co_search_lkh DESTINATION lib/${PROJECT_NAME})
install(TARGETS search_engine DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/co_search_executor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
